\begin{frame}{Producer's optimization problem}
    \begin{wideitemize}
        \item Farmer$i$'s maximizes his CRRA utility $U(W_{i})=W_{i}^{1-\gamma}/(1-\gamma)$, where $W_{i}$ is wealth, defined as:
        \begin{equation}
           W_{i}(a)=PY_{i}(a)-C(a)-\Pi_{i}(\mu_{i}(a))+I_{i}(Y_{i};\mu_{i}(a),\theta,P) 
        \end{equation}
        \item Wealth is equal to revenue ($PY_{i}(a)$) minus cost ($C(a)$) minus producer premium ($\Pi_{i}(\mu_{i}(a)$) plus indemnity paid ($I_{i}(Y_{i};\mu_{i}(a),\theta,P)$).
        \item During \textbf{planting} time, harvest price and yields are not known, so they must be replaced with their expectations, so that premium and indemnity depend on these expected magnitudes.
    \end{wideitemize}
\end{frame}

\begin{frame}{Expected indemnity and premium}
    \begin{wideitemize}
        \item Expected idemnity: $EI(c,a,\theta,s)=P_{c}\mu_{c}(a,s)S(\theta,G_{a,s})$.
        \item Expected premium: $\Pi^{prod}(c,a,\theta,s)=(1-\zeta(\theta))K(\theta)\mu_{c}(a,s)^{1+\beta}$.
        \item $\beta\in(0,1)$, $K(\theta)>0$, and $\zeta(\theta)\in [0,1)$ are the CRF elasticity, rating scale, and subsidy rate.
        \item By Jensen's inequality, we have that:
        \begin{eqnarray}
        E\left[U(W_{i})\right] &=& E\left[W_{i}^{1-\gamma}/(1-\gamma)\right] \geq E(W_{i})^{1-\gamma}/(1-\gamma)\\
        E(W_{i}) &=& E(PY_{i}(a))-C(a)-\Pi^{prod}(c,a,\theta,s)+EI(c,a,\theta,s)
        \end{eqnarray}
    \end{wideitemize}
\end{frame}

\begin{frame}{Revenue breakdown}

    \begin{wideitemize}
        \item Revenue is a function of expected harvest price and yields, expected yields are themselves a function of crop choices $a_{t}$ and weather scenarios, $s_{t}$, so that:
        \begin{equation}
           E(PY_{i}(a)) = P^{e}\mu_{i}(a)
        \end{equation}
        \item $P^{e}$ can be estimated using forward contracts.
        \item $\mu_{i}(a) \equiv \overline{y}(a)$ is the mean yield as a function of rotation choice.
        \item $a_{t}=A(h_{t-1})$, where:
        \begin{equation}
            A(h)=1+\sum_{k=1}^{5}\mathbf{1}_{h_{k}\neq h_{k+1}} \in \{1,2,\dots,6\}
        \end{equation}
    \end{wideitemize}
\end{frame}  

\begin{frame}{Costs}
    \begin{wideitemize}
        \item Both corn and soy have their distinct per-acre cost functions $C^{crop}(corn)$ and $C^{crop}(soy)$.
        \item Similarly, switching from one practice to another has an additional ``penalty'', a.k.a. adjustment cost:
        \begin{equation}
             C^{RCI}(a_{t-1},a_{t})=\kappa\times \max\{0,a_{t}-a_{t-1}\}
        \end{equation}
        \item $\kappa$ is a fixed parameter, but may depend on the geographical location, since RCI tends to be higher in less productive areas, such as far western South Dakota or Northern Wisconsin.
    \end{wideitemize}
\end{frame}

\begin{frame}{Profit (wealth) function}
    \begin{wideitemize}
        \item Given $(h_{t},c_{t},\theta_{t})$, the one-period expected cash flow is:
        \begin{eqnarray}
             \pi(h_{t},c_{t},\theta_{t}) &=& \sum_{s\in \mathcal{S}}\pi_{s}\left[P_{c_{t}}^{e}\mu_{c_{t}}(a_{t},s)+P_{c_{t}}^{e}\mu_{c_{t}}(a_{t},s)S(\theta_{t};G_{a,s})\right.\notag\\ 
             &-& \left. (1-\zeta(\theta_{t}))K(\theta_{t})\mu_{c_{t}}(a_{t},s)^{1+\beta}-C^{crop}(c_{t})-C^{RCI}(a_{t-1},a_{t})\right]
        \end{eqnarray}
        \item $(c_{t}, \theta_{t})\in \{corn, soy\}\times \Theta$, where $\Theta\subset (0,1)$.
        \item The state variable is the six-character strings over $\{C,S\}$, hence:
        \begin{equation}
            \mathcal{T}(h_{t},c_{t})=suffix_{5}(h_{t})||lead(c_{t})=h_{t+1}
        \end{equation}
        \item Control variables are the crop choices ($c_{t}$) and coverage level ($\theta_{t}$)
    \end{wideitemize}
\end{frame}

\begin{frame}{Bellman equation}
    \begin{wideitemize}
        \item Let $\delta = (1+r)^{-1}$ be the discount factor and $V_{T+1}(h)\equiv 0$ be the terminal condition, the Bellman equation will be:
        \begin{equation}
            V_{t}(h_{t}) = \max_{c\in\{corn,soy\},\theta\in\Theta}\pi(h_{t},c_{t},\theta_{t})+\delta E\left[V_{t+1}(h_{t+1})\right],\quad t=T-1,\dots,1
        \end{equation}
        \item Where the expectation is defined across the probabilities $\pi_{s}$.
        \item The state space is, by definition, discrete, hence the solution to this problem consists on finding the optimal value across periods and scenarios.
    \end{wideitemize}
\end{frame}
